# Makefile pour le jeu 2048 avec SDL3
# ===================================

# Compilateur
CXX = g++

# Flags de compilation
CXXFLAGS = -std=c++17 -Wall -Wextra -g -Iinclude -I/mingw64/include -I/mingw64/include/SDL3

# Flags de liaison
LDFLAGS = -L/mingw64/lib -lSDL3 -lSDL3_ttf -lSDL3_image

# Répertoire source
SRC_DIR = src

# Tous les fichiers .cpp dans src/
SRC = $(wildcard $(SRC_DIR)/*.cpp)

# Fichiers objets (.o)
OBJ = $(SRC:.cpp=.o)

# Nom de l'exécutable
TARGET = 2048.exe

# Règle principale : compile tout
all: $(TARGET)

# Lier les fichiers objets pour créer l'exécutable
$(TARGET): $(OBJ)
	$(CXX) -o $@ $^ $(LDFLAGS)

# Compiler chaque fichier .cpp en .o
%.o: %.cpp
	$(CXX) $(CXXFLAGS) -c $< -o $@

# Copier les DLLs SDL3 depuis MSYS2
dll:
	@echo "📋 Copie des DLLs SDL3..."
	cp /mingw64/bin/SDL3.dll . 2>/dev/null || echo "SDL3.dll déjà présente"
	cp /mingw64/bin/SDL3_ttf.dll . 2>/dev/null || echo "SDL3_ttf.dll déjà présente"
	cp /mingw64/bin/SDL3_image.dll . 2>/dev/null || echo "SDL3_image.dll déjà présente"
	@echo "✅ DLLs copiées!"

# Nettoyer les fichiers générés
clean:
	@echo "🧹 Nettoyage..."
	rm -f $(OBJ) $(TARGET)
	@echo "✅ Nettoyage terminé!"

# Compiler et exécuter
run: $(TARGET) dll
	@echo "🚀 Lancement du jeu..."
	./$(TARGET)

# Aide
help:
	@echo "Commandes disponibles:"
	@echo "  make       : Compile le projet"
	@echo "  make run   : Compile et exécute"
	@echo "  make clean : Nettoie les fichiers compilés"
	@echo "  make dll   : Copie les DLLs SDL3"
	@echo "  make help  : Affiche cette aide"

# Indiquer que ces règles ne créent pas de fichiers
.PHONY: all clean run dll help
